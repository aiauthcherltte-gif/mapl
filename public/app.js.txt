'use strict';
const el = id => document.getElementById(id);
const decode = value => Uint8Array.from(atob(value.replace(/-/g, '+').replace(/_/g, '/') + '='.repeat((4 - value.length % 4) % 4)), c => c.charCodeAt(0));
const encode = value => btoa(String.fromCharCode(...new Uint8Array(value))).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
const text = (id, value) => { el(id).textContent = value; };
let busy = false;
let record = null;
let serverMillis = 0;
let receivedAt = 0;
let confirming = null;
let uncertain = true;
let refreshSequence = 0;
let revisionNotice = false;
let actionError = null;
const binding = value => ({revision: value.request.revision, content_hash: value.content_hash});
const matches = (value, expected) => value.request.revision === expected.revision && value.content_hash === expected.content_hash;
const stateKey = value => JSON.stringify([value.request.revision, value.content_hash, value.approval, value.execution]);
const requestId = location.pathname.startsWith('/r/') ? location.pathname.slice(3) : null;

async function api(path, body) {
  const response = await fetch(path, {method: body === undefined ? 'GET' : 'POST',
    headers: body === undefined ? {} : {'Content-Type': 'application/json'},
    body: body === undefined ? undefined : JSON.stringify(body), cache: 'no-store', credentials: 'same-origin'});
  const result = await response.json();
  if (!response.ok) {
    const error = new Error(result.error || 'request_refused');
    error.refusal = true;
    throw error;
  }
  return result;
}
function credentialsOptions(options) {
  options.challenge = decode(options.challenge);
  if (options.user) options.user.id = decode(options.user.id);
  if (options.allowCredentials) options.allowCredentials = options.allowCredentials.map(c => ({...c, id: decode(c.id)}));
  return {publicKey: options};
}
function serialized(credential) {
  const response = credential.response;
  const result = {id: credential.id, type: credential.type, response: {clientDataJSON: encode(response.clientDataJSON)}};
  if (response.attestationObject) result.response.attestationObject = encode(response.attestationObject);
  else Object.assign(result.response, {authenticatorData: encode(response.authenticatorData), signature: encode(response.signature)});
  return result;
}
function summary(value) {
  text('passkey-summary', '활성 패스키 ' + value.credential_count + '개');
}
function render() {
  if (!record) return;
  el('passkey-summary').hidden = true;
  document.querySelector('nav').hidden = true;
  const view = record.view;
  for (const id of ['title', 'change', 'target', 'impact', 'scope', 'failure', 'deadline']) text(id, view[id]);
  // Avoid replacing an unchanged state node, so its mutation history is meaningful.
  const state = uncertain ? '연결 확인 중 · 서버의 현재 상태를 확인하지 못했습니다.' : view.state;
  if (el('state').textContent !== state) text('state', state);
  text('details', view.details_text);
  const pending = record.approval === 'pending' && record.execution === 'not_started';
  el('actions').hidden = !pending || uncertain;
  text('site-host', '주소창과 비교: ' + record.site_host);
  sizeApproval();
  el('withdraw').hidden = uncertain || record.approval !== 'approved' || record.execution !== 'not_started' || Boolean(record.claim_id);
  for (const id of ['approve', 'decline', 'withdraw']) el(id).disabled = busy || uncertain;
}
// Re-measure when host wrapping, fonts or viewport dimensions change.
function sizeApproval() {
  const actions = el('actions');
  if (actions.hidden) return;
  document.querySelector('main').style.setProperty('--approval-height', actions.getBoundingClientRect().height + 'px');
}
if (typeof ResizeObserver !== 'undefined') new ResizeObserver(sizeApproval).observe(el('actions'));
async function refresh() {
  if (!requestId) return;
  const sequence = ++refreshSequence;
  let next;
  try { next = await api('/api/requests/' + requestId); }
  catch (error) {
    if (sequence !== refreshSequence) return;
    uncertain = true; render();
    throw error;
  }
  // A GET started before an action must never overwrite its follow-up read.
  if (sequence !== refreshSequence) return;
  const changed = revisionNotice || (confirming && !matches(next, confirming));
  if (changed) confirming = null;
  record = next;
  uncertain = false;
  serverMillis = Date.parse(record.server_now);
  receivedAt = performance.now();
  if (record.approval !== 'pending') confirming = null;
  if (actionError && actionError.key !== stateKey(record)) actionError = null;
  text('error', changed ? '내용이 바뀌었습니다 · 다시 확인하세요.' : (actionError ? actionError.message : ''));
  revisionNotice = false;
  render();
}
async function decide() {
  if (busy || uncertain || !confirming) return;
  const shown = confirming;
  ++refreshSequence;
  busy = true;
  actionError = null;
  render();
  text('error', '');
  try {
    const options = await api('/api/requests/' + requestId + '/approve/options', shown);
    const credential = await navigator.credentials.get(credentialsOptions(options));
    if (confirming !== shown || uncertain || !matches(record, shown)) throw new Error('revision_changed');
    const payload = {credential: serialized(credential), revision: shown.revision};
    const path = '/api/requests/' + requestId + '/approve/verify';
    if (record.test_scenario === 'double_click') {
      const answers = await Promise.allSettled([api(path, payload), api(path, payload)]);
      if (!answers.some(answer => answer.status === 'fulfilled')) throw answers[0].reason;
    } else await api(path, payload);
    confirming = null;
  } catch (error) {
    // Read server truth before offering another action after a lost response.
    revisionNotice = error.message === 'revision_changed' || error.message === 'superseded';
    if (error.refusal && !revisionNotice) actionError = {key: stateKey(record), message: error.message};
    uncertain = true;
  } finally {
    try { await refresh(); }
    catch (_) { text('error', '연결을 확인하지 못했습니다. 현재 상태를 다시 확인 중입니다.'); }
    busy = false;
    render();
  }
}
el('approve').addEventListener('click', () => {
  if (busy || uncertain || !record || record.approval !== 'pending') return;
  confirming = Object.freeze(binding(record)); actionError = null; text('error', ''); return decide();
});

for (const action of ['decline', 'withdraw']) el(action).addEventListener('click', async () => {
  if (busy || uncertain) return;
  ++refreshSequence;
  busy = true;
  actionError = null;
  render();
  try { await api('/api/requests/' + requestId + '/' + action, {}); }
  catch (_) { uncertain = true; }
  finally {
    try { await refresh(); }
    catch (_) { text('error', '연결을 확인하지 못했습니다. 현재 상태를 다시 확인 중입니다.'); }
    busy = false; render();
  }
});
el('register').addEventListener('click', async () => {
  if (busy) return;
  busy = true;
  el('register').disabled = true;
  try {
    const code = el('setup-code').value.trim();
    const options = await api('/api/register/options', {code});
    const credential = await navigator.credentials.create(credentialsOptions(options));
    const result = await api('/api/register/verify', {code, credential: serialized(credential)});
    text('registration-status', '시험용 패스키 1개 등록됨 · ' + result.registered_at_text);
    el('setup-code').value = '';
    el('setup-code').disabled = true;
  } catch (error) { text('error', error.message || error.name); el('register').disabled = false; }
  finally { busy = false; }
});
let adding = null;
el('login').addEventListener('click', async () => {
  if (busy) return; busy = true; el('login').disabled = true;
  try {
    const options = await api('/api/login/options', {});
    const credential = await navigator.credentials.get(credentialsOptions(options));
    const challenge = encode(options.challenge);
    const result = await api('/api/login/verify', {challenge, credential: serialized(credential)});
    summary(result); location.assign(/^#\/r\/[0-9a-f]{32}$/.test(location.hash) ? location.hash.slice(1) : '/');
  } catch (error) { text('error', error.message || error.name); }
  finally { busy = false; el('login').disabled = false; }
});
el('logout').addEventListener('click', async () => {
  try { const options = await api('/api/logout/options', {}); const credential = await navigator.credentials.get(credentialsOptions({...options})); await api('/api/logout', {action_id: options.action_id, credential: serialized(credential)}); location.assign('/login'); }
  catch (error) { text('error', error.message); }
});
el('add-create').addEventListener('click', async () => {
  if (busy) return; busy = true; el('add-create').disabled = true;
  try {
    const action = await api('/api/credential-add/options', {});
    const credential = await navigator.credentials.create(credentialsOptions(action.registration));
    adding = await api('/api/credential-add/prepare', {action_id: action.action_id, credential: serialized(credential)});
    text('add-status', '새 패스키 추가 승인 대기 · 기존 패스키로 확인하세요.');
    el('add-approve').hidden = false;
  } catch (error) { text('error', error.message || error.name); }
  finally { busy = false; el('add-create').disabled = false; }
});
el('add-approve').addEventListener('click', async () => {
  if (busy || !adding) return; busy = true; el('add-approve').disabled = true;
  try {
    const action = adding;
    const credential = await navigator.credentials.get(credentialsOptions({...action}));
    const result = await api('/api/credential-add/verify', {action_id: action.action_id, credential: serialized(credential)});
    summary(result); adding = null; el('add-approve').hidden = true;
    text('add-status', '시험용 패스키 추가됨 · 개수와 마지막 추가 시각을 확인하세요.');
  } catch (error) { text('error', error.message || error.name); }
  finally { busy = false; el('add-approve').disabled = false; }
});
if (location.pathname === '/login') el('login-panel').hidden = false;
else if (location.pathname === '/') {
  el('list-panel').hidden = false;
  api('/api/requests').then(result => {
    summary(result);
    text('runner-summary', (result.last_response ? '실행기 마지막 응답 ' + result.last_response + ' · ' : '') + '켜짐·꺼짐은 이 정보만으로 알 수 없습니다');
    for (const r of result.requests) {
      const link = document.createElement('a'), row = document.createElement('p');
      link.href = '/r/' + r.request.request_id;
      for (const value of [r.list_title, r.list_meta, r.display_state]) {
        const line = document.createElement('span'); line.textContent = value; link.appendChild(line);
      }
      row.appendChild(link); el('request-list').appendChild(row);
    }
  }).catch(error => text('error', error.message));
} else if (location.pathname === '/history') {
  el('history-panel').hidden = false;
  api('/api/history').then(result => text('auth-history', JSON.stringify(result.history, null, 2))).catch(error => text('error', error.message));
} else if (location.pathname === '/credential-add') {
  el('add-panel').hidden = false;
  api('/api/requests').then(summary).catch(error => text('error', error.message));
} else if (location.pathname === '/register') el('registration').hidden = false;
else {
  el('request').hidden = false;
  el('request').insertBefore(el('band'), el('details-panel'));
  refresh().catch(() => text('error', '연결을 확인하지 못했습니다.'));
  const active = () => record && (['claimed', 'running', 'checking', 'recovering'].includes(record.execution) || (record.approval === 'approved' && record.execution === 'not_started'));
  setInterval(() => { if (!busy && !document.hidden && (uncertain || active())) refresh().catch(() => text('error', '연결을 확인하지 못했습니다.')); }, 5000);
  document.addEventListener('visibilitychange', () => { if (!document.hidden && !busy) refresh().catch(() => {}); });
  setInterval(() => {
    if (!record) return;
    const now = serverMillis + performance.now() - receivedAt;
    const expires = Date.parse(record.request.expires_at);
    const minutes = Math.max(0, Math.floor((expires - now) / 60000));
    text('deadline', record.view.deadline.replace(/\(\d+분 남음\)/, '(' + minutes + '분 남음)'));
  }, 1000);
}
